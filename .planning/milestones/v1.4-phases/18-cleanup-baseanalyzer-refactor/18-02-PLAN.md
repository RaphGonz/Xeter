class BaseAnalyzer(ABC):
    """Generic root: embedding helpers shared by span and trace analyzers."""
    def __init__(self, embedder: EmbedderClient, thresholds: dict[str, float]) -> None: ...
    def embed(self, text: str) -> np.ndarray: ...
    def compare(self, a, b) -> float: ...
    def log_score(self, metric_name, score) -> None: ...
    def flush_scores(self) -> list[...]: ...
    @property
    @abstractmethod
    def name(self) -> str: ...
    # NO analyze() abstract method in root


class BaseSpanAnalyzer(BaseAnalyzer):
    """Analyzer contract for single-span analysis."""
    @abstractmethod

class BaseTraceAnalyzer(BaseAnalyzer):
    """Analyzer contract for full-trace analysis (list of spans)."""
