BRPOP worker loop (from 03-RESEARCH.md — use BRPOP not BLPOP for FIFO with LPUSH):
```python
# queue.py uses LPUSH; BRPOP (right-pop) gives FIFO order — per queue.py docstring
import signal, redis, os, logging

QUEUE_KEY = "analysis_queue"
BRPOP_TIMEOUT = 2  # seconds — allows SIGTERM response within 2s

running = True
def _handle_signal(signum, frame):
    global running
    running = False

signal.signal(signal.SIGTERM, _handle_signal)
signal.signal(signal.SIGINT, _handle_signal)

r = redis.Redis.from_url(os.environ["REDIS_URL"], decode_responses=True)

while running:
    result = r.brpop(QUEUE_KEY, timeout=BRPOP_TIMEOUT)
    if result is None:
        continue
    _, span_id = result
    try:
        process_span(span_id)
    except Exception as exc:
        logger.error("worker: failed to process span %s: %s", span_id, exc)
        # log-and-skip: no retry, no dead-letter queue (user decision)
```

process_span dispatch pattern (from 03-CONTEXT.md locked decision):
```python
def process_span(span_id: str) -> None:
    span = fetch_span(span_id)
    all_flags, all_scores = [], []
    for analyzer in ANALYZERS:
        flags = analyzer.analyze(span)
        scores = analyzer.flush_scores()
        all_flags.extend(flags)
        all_scores.extend(scores)
    write_scores(span_id, span.tenant_id, all_scores)
    if all_flags:
        write_flags(span_id, span.tenant_id, span.trace_id, all_flags)
```
