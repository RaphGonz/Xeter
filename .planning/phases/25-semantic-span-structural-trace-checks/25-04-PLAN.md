    _check_stale_context(self, spans: list[SpanData]) -> list[Flag]:
    - Guard `if len(spans) < 2: return []`.
    - Loop i from 1 to len(spans)-1.
    - Skip pair if spans[i-1].tool_output is None or spans[i].prompt is None (no log_score — guard).
    - score = fuzz.ratio(spans[i].prompt, spans[i-1].tool_output).
    - self.log_score("stale_context", score).
    - if score >= self._thresholds["stale_context"]: append Flag(flag_type="stale_context", score=score, detail={"metric": "stale_context", "span_index": i, "low_confidence": True}).
    - return flags.

    _check_step_repetition(self, spans: list[SpanData]) -> list[Flag]:
    - Guard `if len(spans) < 2: return []`.
    - For every unique pair (a, b) where a < b (use nested loops or itertools.combinations).
    - Skip if spans[a].tool_name is None or spans[b].tool_name is None.
    - key_a = f"{spans[a].tool_name} {spans[a].tool_arguments or ''}".strip().
    - key_b = f"{spans[b].tool_name} {spans[b].tool_arguments or ''}".strip().
    - score = fuzz.token_sort_ratio(key_a, key_b).
    - self.log_score("step_repetition", score).
    - if score >= self._thresholds["step_repetition"]: append Flag(flag_type="step_repetition", score=score, detail={"metric": "step_repetition", "span_index_a": a, "span_index_b": b}).
    - return flags.

    _check_termination_loop(self, spans: list[SpanData]) -> list[Flag]:
    - Guard `if len(spans) < 2: return []`.
    - n = int(self._thresholds["termination_loop_n"]).
    - Track current_tool = None, run_length = 0, flagged_tools = set().
    - For each span in spans:
      - If span.tool_name is None: reset current_tool = None, run_length = 0; continue.
      - If span.tool_name == current_tool: run_length += 1.
      - Else: current_tool = span.tool_name; run_length = 1.
      - If run_length >= n and current_tool not in flagged_tools: append Flag; flagged_tools.add(current_tool).
    - return flags.
    - No log_score for termination_loop (count-based, not similarity-based — no calibration threshold to sweep; termination_loop_n is an integer count).

    _check_context_propagation_failure(self, spans: list[SpanData]) -> list[Flag]:
    - Guard `if len(spans) < 2: return []`.
    - Loop i from 1 to len(spans)-1.
    - Skip if spans[i-1].tool_output is None or spans[i].prompt is None.
